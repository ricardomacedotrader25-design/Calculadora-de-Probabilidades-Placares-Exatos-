<script>
  const fmtPct = n => n.toLocaleString('pt-BR',{minimumFractionDigits:2,maximumFractionDigits:2}) + '%';
  const fmtNum = n => parseFloat(n.toFixed(1)).toString();
  const parseVal = v => { const s=String(v??'').trim().replace(',', '.'); const x=parseFloat(s); return (isFinite(x)&&x>0)?x:NaN; };

  function calcProbFromRow(tr){
    const v = parseVal(tr.querySelector('input[data-type="back"]').value);
    if (isNaN(v)) return null; 
    return 100 / v;
  }

  /* Utilitários para interpolar cores HEX ←→ RGB */
  function hexToRgb(hex){
    const h = hex.replace('#','');
    const b = h.length===3 ? h.split('').map(c=>c+c).join('') : h;
    return { r: parseInt(b.slice(0,2),16), g: parseInt(b.slice(2,4),16), b: parseInt(b.slice(4,6),16) };
  }
  function rgbToHex({r,g,b}){
    const toHex = v => v.toString(16).padStart(2,'0');
    return `#${toHex(r)}${toHex(g)}${toHex(b)}`.toUpperCase();
  }
  function mixHex(aHex,bHex,t){
    const a=hexToRgb(aHex), b=hexToRgb(bHex);
    const lerp=(x,y,t)=>Math.round(x+(y-x)*t);
    return rgbToHex({r:lerp(a.r,b.r,t), g:lerp(a.g,b.g,t), b:lerp(a.b,b.b,t)});
  }

  function paintGradient(){
    const cells=[...document.querySelectorAll('#cs tbody td.prob')];
    const vals=cells.map(td=>parseFloat(td.textContent.replace('%','').replace(',','.'))).filter(v=>!isNaN(v));
    if(!vals.length){cells.forEach(td=>{td.style.background=''; td.style.color='';}); return;}

    const min=Math.min(...vals), max=Math.max(...vals), span=(max-min)||1;
    const lowBG=getComputedStyle(document.documentElement).getPropertyValue('--prob-low-bg').trim() || '#FDE8E8';
    const lowFG=getComputedStyle(document.documentElement).getPropertyValue('--prob-low-fg').trim() || '#B5261B';
    const highBG=getComputedStyle(document.documentElement).getPropertyValue('--prob-high-bg').trim() || '#CFF9E9';
    const highFG=getComputedStyle(document.documentElement).getPropertyValue('--prob-high-fg').trim() || '#0E7F62';

    cells.forEach(td=>{
      const v=parseFloat(td.textContent.replace('%','').replace(',','.'));
      if(isNaN(v)){ td.style.background=''; td.style.color=''; return; }
      let t=(v-min)/span;       // 0 → menor prob | 1 → maior prob
      t = Math.pow(t, 0.65);    // easing p/ puxar mais o verde no topo
      td.style.background = mixHex(lowBG, highBG, t);
      td.style.color      = mixHex(lowFG, highFG, t);
    });
  }

  function recalcAll(){
    document.querySelectorAll('#cs tbody tr').forEach((tr,i)=>{
      if(!tr.dataset.index) tr.dataset.index=i;
      const p = calcProbFromRow(tr);
      tr.querySelector('.prob').textContent = (p===null)?'—':fmtPct(p);
      tr.dataset.prob = (p===null)?'':p.toFixed(6);
    });
    paintGradient();
  }

  function sortByProb(dir){
    const tbody=document.querySelector('#cs tbody');
    const rows=[...tbody.querySelectorAll('tr')];
    rows.sort((a,b)=>{
      const pa=a.dataset.prob===''?-Infinity:parseFloat(a.dataset.prob);
      const pb=b.dataset.prob===''?-Infinity:parseFloat(b.dataset.prob);
      return dir==='desc'?(pb-pa):(pa-pb);
    });
    rows.forEach(r=>tbody.appendChild(r));
    paintGradient();
  }
  function sortOriginal(){
    const tbody=document.querySelector('#cs tbody');
    const rows=[...tbody.querySelectorAll('tr')];
    rows.sort((a,b)=>parseInt(a.dataset.index)-parseInt(b.dataset.index));
    rows.forEach(r=>tbody.appendChild(r));
    paintGradient();
  }

  function applyBulkOdds() {
    const oddsArray = document.getElementById('bulkOddsBack').value
      .split(/[, ]+/).map(parseVal).filter(v => !isNaN(v));
    const backOddsInputs = document.querySelectorAll('#cs input[data-type="back"]');
    backOddsInputs.forEach((input, index) => input.value = index < oddsArray.length ? fmtNum(oddsArray[index]) : '');
    recalcAll();
  }
  function clearAllBackOdds() {
    document.querySelectorAll('#cs input[data-type="back"]').forEach(input => input.value = '');
    const bulk = document.getElementById('bulkOddsBack'); if (bulk) bulk.value = '';
    recalcAll();
  }
  function exportCSV(){
    const out=[];
    out.push(['Placares','Odd Back','Probabilidade %'].join(','));
    document.querySelectorAll('#cs tbody tr').forEach(tr=>{
      const placar=tr.children[0].textContent.trim();
      const back=tr.querySelector('input[data-type="back"]').value.trim();
      const prob=tr.querySelector('.prob').textContent.trim();
      out.push([placar,back,prob].map(v=>{
        const s=v.replaceAll('"','"'); return /,|\n|"/.test(s)?`"${s}"`:s;
      }).join(','));
    });
    const blob=new Blob([out.join('\n')],{type:'text/csv;charset=utf-8;'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a'); a.href=url; a.download='correct-score.csv';
    document.body.appendChild(a); a.click(); a.remove(); URL.revokeObjectURL(url);
  }

  // Listeners
  document.addEventListener('input', e=>{
    if(e.target && e.target.matches('#cs input.odds')) recalcAll();
  });
  document.getElementById('sortDesc').addEventListener('click',()=>{recalcAll();sortByProb('desc');});
  document.getElementById('sortAsc').addEventListener('click', ()=>{recalcAll();sortByProb('asc');});
  document.getElementById('sortOrig').addEventListener('click',()=>{recalcAll();sortOriginal();});
  document.getElementById('applyBulkOdds').addEventListener('click', applyBulkOdds);
  document.getElementById('clearOdds').addEventListener('click', clearAllBackOdds);
  document.getElementById('exportCSV').addEventListener('click',exportCSV);

  // Estado inicial
  document.addEventListener('DOMContentLoaded', () => {
    recalcAll();
  });
</script>
